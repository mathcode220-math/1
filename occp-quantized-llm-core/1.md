```yaml
    `timescale 1ns/1ps

    module loopback_register #(
        parameter int DATA_WIDTH = 16,
        parameter int VECTOR_LEN = 512
    )(
        input  logic                    clk,
        input  logic                    rst_n,
        input  logic                    en,
        input  logic [DATA_WIDTH-1:0]   data_in  [0:VECTOR_LEN-1],
        input  logic                    valid_in,
        output logic [DATA_WIDTH-1:0]   data_out [0:VECTOR_LEN-1],
        output logic                    valid_out
    );

        always_ff @(posedge clk or negedge rst_n) begin
            if (!rst_n) begin
                valid_out <= 1'b0;
                for (int i = 0; i < VECTOR_LEN; i++) data_out[i] <= '0;
            end else if (en) begin
                valid_out <= valid_in;
                for (int i = 0; i < VECTOR_LEN; i++) data_out[i] <= data_in[i];
            end
        end

    endmodule

  validation:
    - "verilator --lint-only -Wno-fatal rtl/recurrent/loopback_register.sv"
```

```yaml
  file_path: "rtl/recurrent/weight_pingpong.sv"
  content: |
    `timescale 1ns/1ps

    module weight_pingpong #(
        parameter int DATA_WIDTH   = 16,
        parameter int WEIGHT_DEPTH = 4096
    )(
        input  logic                                clk,
        input  logic                                rst_n,
        input  logic                                buf_sel,
        input  logic                                load_en,
        input  logic [DATA_WIDTH-1:0]               load_data,
        input  logic [$clog2(WEIGHT_DEPTH)-1:0]     load_addr,
        input  logic [$clog2(WEIGHT_DEPTH)-1:0]     read_addr,
        output logic [DATA_WIDTH-1:0]               weight_out
    );

        logic [DATA_WIDTH-1:0] buf_a [0:WEIGHT_DEPTH-1];
        logic [DATA_WIDTH-1:0] buf_b [0:WEIGHT_DEPTH-1];

        always_ff @(posedge clk) begin
            if (load_en) begin
                if (buf_sel)
                    buf_a[load_addr] <= load_data;
                else
                    buf_b[load_addr] <= load_data;
            end
        end

        assign weight_out = buf_sel ? buf_b[read_addr] : buf_a[read_addr];

    endmodule

  validation:
    - "verilator --lint-only -Wno-fatal rtl/recurrent/weight_pingpong.sv"
```

```yaml
    )(
        input  logic                        clk,
        input  logic                        rst_n,
        input  logic                        start,
        input  logic [LATENCY_WIDTH-1:0]    latency_cfg,
        output logic                        done
    );

        logic [LATENCY_WIDTH-1:0] counter;
        logic                     active;

        always_ff @(posedge clk or negedge rst_n) begin
            if (!rst_n) begin
                counter <= '0;
                active  <= 1'b0;
                done    <= 1'b0;
            end else begin
                done <= 1'b0;
                if (start) begin
                    counter <= latency_cfg - 1'b1;
                    active  <= 1'b1;
                end else if (active) begin
                    if (counter == '0) begin
                        done   <= 1'b1;
                        active <= 1'b0;
                    end else begin
                        counter <= counter - 1'b1;
                    end
                end
            end
        end

    endmodule

  validation:
    - "verilator --lint-only -Wno-fatal rtl/recurrent/sa_done_generator.sv"
```

### 📄 الملف 5.3: `rtl/control/occp_pipeline_fsm.sv`

```yaml
TASK_5.3_CREATE_PIPELINE_FSM:
  file_path: "rtl/control/occp_pipeline_fsm.sv"
  content: |
    `timescale 1ns/1ps
    import model_metadata_pkg::*;

    module occp_pipeline_fsm #(
        parameter int MAX_LAYERS = 128
    )(
        input  logic                clk,
        input  logic                rst_n,
        input  logic                host_start,
        output logic                host_busy,
        input  model_header_t       metadata_in,
        input  logic                metadata_valid,
        output logic                sel_external,
        output logic                capture_loop,
        output logic                start_layer,
        input  logic                layer_done,
        input  logic                softmax_done,
        output logic                buf_sel,
        output logic                weight_load_en,
        output logic [7:0]          weight_layer_idx,
        input  logic                weight_ready,
        output logic                output_token_valid,
        output logic [7:0]          layer_counter_out
    );

        fsm_state_t state, next_state;
        logic [7:0]  layer_counter;
        logic [7:0]  total_layers;
        logic [7:0]  layer_after_compute;
        logic        is_last_layer;

        always_comb begin
            next_state = state;
            unique case (state)
                S_IDLE: begin
                    if (host_start && metadata_valid) next_state = S_LOAD_META;
                end
                S_LOAD_META:    next_state = S_PREFETCH_W;
                S_PREFETCH_W:   if (weight_ready) next_state = S_LOAD_LAYER_0;
                S_LOAD_LAYER_0: next_state = S_COMPUTE;
                S_COMPUTE:      next_state = S_WAIT_DONE;
                S_WAIT_DONE:    if (layer_done && softmax_done) next_state = S_LOOP_CHECK;
                S_LOOP_CHECK:   next_state = is_last_layer ? S_OUTPUT_TOKEN : S_COMPUTE;
                S_OUTPUT_TOKEN: if (!host_start) next_state = S_IDLE;
                S_ERROR:        next_state = S_IDLE;
                default:        next_state = S_IDLE;
            endcase
        end

        always_ff @(posedge clk or negedge rst_n) begin
            if (!rst_n) state <= S_IDLE;
            else        state <= next_state;
        end

        always_ff @(posedge clk or negedge rst_n) begin
            if (!rst_n) begin
                total_layers        <= '0;
                layer_counter       <= '0;
                layer_after_compute <= '0;
            end else begin
                case (state)
                    S_LOAD_META: begin
                        total_layers  <= metadata_in.total_layers;
                        layer_counter <= 8'd0;
                    end
                    S_WAIT_DONE: begin
                        if (layer_done && softmax_done) begin
                            layer_counter       <= layer_counter + 1'b1;
                            layer_after_compute <= layer_counter + 8'd2;
                        end
                    end
                    S_OUTPUT_TOKEN: begin
                        if (!host_start) layer_counter <= 8'd0;
                    end
                endcase
            end
        end

        assign is_last_layer = (layer_counter >= total_layers - 1'b1);

        always_comb begin
            sel_external       = 1'b0;
            capture_loop       = 1'b0;
            start_layer        = 1'b0;
            buf_sel            = 1'b0;
            weight_load_en     = 1'b0;
            weight_layer_idx   = 8'd0;
            output_token_valid = 1'b0;
            host_busy          = 1'b1;

            unique case (state)
                S_IDLE: begin
                    host_busy    = 1'b0;
                    sel_external = 1'b1;
                end
                S_PREFETCH_W: begin
                    buf_sel          = 1'b0;
                    weight_load_en   = 1'b1;
                    weight_layer_idx = 8'd0;
                end
                S_LOAD_LAYER_0: begin
                    sel_external     = 1'b1;
                    buf_sel          = 1'b0;
                    weight_load_en   = 1'b1;
                    weight_layer_idx = layer_after_compute;
                end
                S_COMPUTE: begin
                    start_layer      = 1'b1;
                    sel_external     = (layer_counter == 8'd0);
                    capture_loop     = (layer_counter != 8'd0);
                    buf_sel          = layer_counter[0];
                    weight_load_en   = 1'b1;
                    weight_layer_idx = layer_after_compute;
                end
                S_WAIT_DONE: begin
                    capture_loop     = (layer_counter != 8'd0);
                    buf_sel          = layer_counter[0];
                end
                S_LOOP_CHECK: begin
                    capture_loop = 1'b1;
                end
                S_OUTPUT_TOKEN: begin
                    output_token_valid = 1'b1;
                    host_busy          = 1'b0;
                end
            endcase
        end

        assign layer_counter_out = layer_counter;

    endmodule
```

```yaml
TASK_6.1_CREATE_RECURRENT_DATAPATH:
  file_path: "rtl/top/recurrent_datapath.sv"
  content: |
    `timescale 1ns/1ps

    module recurrent_datapath #(
        parameter int DATA_WIDTH = 16,
        parameter int VECTOR_LEN = 512,
        parameter int LAYER_LATENCY = 3
    )(
        input  logic                         clk,
        input  logic                         rst_n,
        input  logic                         sel_external,
        input  logic                         capture_loop,
        input  logic                         start_layer,
        output logic                         layer_done,
        input  logic [DATA_WIDTH-1:0]        ext_data  [0:VECTOR_LEN-1],
        input  logic                         ext_valid,
        output logic [DATA_WIDTH-1:0]        final_data [0:VECTOR_LEN-1],
        output logic                         final_valid
    );

        logic [DATA_WIDTH-1:0] mux_data  [0:VECTOR_LEN-1];
        logic                  mux_valid;
        logic [DATA_WIDTH-1:0] layer_out [0:VECTOR_LEN-1];
        logic                  layer_valid;
        logic [DATA_WIDTH-1:0] loop_data [0:VECTOR_LEN-1];
        logic                  loop_valid;

        layer_loopback_mux #(
            .DATA_WIDTH(DATA_WIDTH),
            .VECTOR_LEN(VECTOR_LEN)
        ) u_mux (
            .sel_external (sel_external),
            .ext_data     (ext_data),
            .ext_valid    (ext_valid),
            .loop_data    (loop_data),
            .loop_valid   (loop_valid),
            .mux_data     (mux_data),
            .mux_valid    (mux_valid)
        );

        // Layer stub — للتكامل، يُستبدل بـ systolic_array_param لاحقاً
        localparam int CNT_W = $clog2(LAYER_LATENCY + 1);
        logic [CNT_W-1:0] delay_cnt;
        logic             computing;

        always_ff @(posedge clk or negedge rst_n) begin
            if (!rst_n) begin
                delay_cnt   <= '0;
                computing   <= 1'b0;
                layer_valid <= 1'b0;
                for (int i = 0; i < VECTOR_LEN; i++) layer_out[i] <= '0;
            end else begin
                if (start_layer && mux_valid && !computing) begin
                    delay_cnt <= LAYER_LATENCY[CNT_W-1:0];
                    computing <= 1'b1;
                    layer_valid <= 1'b0;
                end else if (computing) begin
                    if (delay_cnt == 1) begin
                        layer_out   <= mux_data;
                        layer_valid <= 1'b1;
                        computing   <= 1'b0;
                    end
                    delay_cnt <= delay_cnt - 1'b1;
                end else begin
                    layer_valid <= 1'b0;
                end
            end
        end

        assign layer_done = computing && (delay_cnt == 1);

        loopback_register #(
            .DATA_WIDTH(DATA_WIDTH),
            .VECTOR_LEN(VECTOR_LEN)
        ) u_loop_reg (
            .clk       (clk),
            .rst_n     (rst_n),
            .en        (capture_loop),
            .data_in   (layer_out),
            .valid_in  (layer_valid),
            .data_out  (loop_data),
            .valid_out (loop_valid)
```

```yaml
TASK_8.1_CREATE_METADATA_WRITER:
  file_path: "software/metadata_writer.py"
  content: |
    import struct
    from dataclasses import dataclass
    from enum import IntEnum
    import numpy as np


    class ActivationKind(IntEnum):
        RELU = 0
        GELU = 1
        SILU = 2
        TANH = 3
        SOFTMAX = 4


    HEADER_FORMAT = "<BI BH B"
    HEADER_SIZE = struct.calcsize(HEADER_FORMAT)


    @dataclass
    class ModelMetadata:
        total_layers: int
        layer_weight_bytes: int
        activation_kind: ActivationKind
        vector_len: int
        reserved: int = 0

        def pack(self) -> bytes:
            return struct.pack(HEADER_FORMAT, self.total_layers,
                               self.layer_weight_bytes, self.activation_kind,
                               self.vector_len, self.reserved)

        @classmethod
        def unpack(cls, data: bytes) -> "ModelMetadata":
            t, w, a, v, r = struct.unpack(HEADER_FORMAT, data[:HEADER_SIZE])
            return cls(t, w, ActivationKind(a), v, r)


    def export_with_header(output_path, layers, activation_kind=ActivationKind.RELU,
                           vector_len=512):
        if not layers:
            raise ValueError("Empty layer list")
        layer_bytes = [l.astype(np.int8).tobytes() for l in layers]
        sizes = [len(b) for b in layer_bytes]
        if len(set(sizes)) != 1:
            raise ValueError(f"Layer sizes differ: {sizes}")
        meta = ModelMetadata(len(layers), sizes[0], activation_kind, vector_len)
        with open(output_path, "wb") as f:
            f.write(meta.pack())
            for b in layer_bytes:
                f.write(b)
        print(f"✅ Exported: {output_path}")
        print(f"   Layers: {meta.total_layers}, Size/layer: {meta.layer_weight_bytes}")


    if __name__ == "__main__":
        layers = [np.random.randint(-127, 127, 16, dtype=np.int8) for _ in range(4)]
        export_with_header("/tmp/test_model.bin", layers)
```

```yaml
    import sys
    from pathlib import Path
    from metadata_writer import ModelMetadata, HEADER_SIZE


    def validate_bin(bin_path):
        p = Path(bin_path)
        if not p.exists():
            print(f"❌ File not found: {bin_path}")
            return False
        with open(bin_path, "rb") as f:
            header_bytes = f.read(HEADER_SIZE)
            payload = f.read()
        meta = ModelMetadata.unpack(header_bytes)
        expected = meta.total_layers * meta.layer_weight_bytes
        print(f"Layers: {meta.total_layers}")
        print(f"Bytes/layer: {meta.layer_weight_bytes}")
        print(f"Activation: {meta.activation_kind.name}")
        print(f"Vector len: {meta.vector_len}")
        print(f"Payload expected: {expected}, actual: {len(payload)}")
        return expected == len(payload)


    if __name__ == "__main__":
        if len(sys.argv) < 2:
            print("Usage: python validate_bin.py <file.bin>")
            sys.exit(1)
```
